265    /**
266     * Resolve flagged create/drop pairs by DECLARING the rename on the
267     * blueprint — the decision, not a guess.
268     *
269     * When the user confirms, the desired blueprint for the new table gets
270     * a renamedFrom declaration; the caller re-plans so the differ verifies
271     * the declaration and emits the real RenameTable change. Each old table
272     * can only be claimed by one rename (the differ pairs every create with
273     * its best-overlap drop, so multiple creates can claim the same drop —
274     * first claim wins).
275     *
276     * With --force every suggestion is auto-accepted. Without a TTY or an
277     * injected input stream the flagged pair is kept as-is (the plan shows
278     * the POSSIBLE RENAME annotation and the destructive gate handles it)
279     * — never block on stdin we cannot read.
280     *
281     * @param list<SchemaChange> $changes The current plan
282     * @param list<Blueprint> $pristine The desired blueprints (mutated in place)
283     * @param bool $force Auto-accept every suggestion
284     * @return bool Whether any rename was declared (caller must re-plan)
285     */

289        $pairs = []; // createTable => dropTable
290        foreach ($changes as $change) {
291            if ($change->operation === SchemaOperation::CreateTable && $change->renameOf !== null) {
292                $pairs[$change->table] = $change->renameOf;
293            }
294        }
295
296        if ($pairs === []) {
297            return false;
298        }
299
300        // Nothing to prompt with (no TTY, no injected stream) and no
301        // --force: keep the flagged pair as-is.
302        if (!$force && !self::canPrompt()) {
303            return false;
304        }
305
306        $declared = false;
307        $claimedOldTables = []; // each old table can only be renamed once
308
309        foreach ($pairs as $newTable => $oldTable) {
310            // Only the first claim on an old table becomes a rename.
311            if (in_array($oldTable, $claimedOldTables, true)) {
312                continue;
313            }

315            // readAnswer() already normalizes to a bool — a plain falsy
316            // check is the correct gate (a strict string compare would
317            // treat a typed "yes" as a decline and re-prompt every
318            // remaining pair with the same old table).
319            if ($force) {

323                    "POSSIBLE RENAME: [{$oldTable}] → [{$newTable}]. Treat as a rename?"
324                );

326            }

328            if (!$answer) {

330            }

333                if ($blueprint->getTable() === $newTable) {
334                    $pristine[$index] = $blueprint->renamedFrom($oldTable);
335                    $claimedOldTables[] = $oldTable;
336                    $declared = true;
337                    break;
338                }
339            }
340        }
