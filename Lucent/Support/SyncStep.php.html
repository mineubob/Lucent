9/**
10 * A hand-authored callable sync step — the PHP sibling of a `--step=.sql`
11 * file. Where an SQL step carries its transform as reviewable statement
12 * text (the dry run prints every statement), a callable step carries it as
13 * runtime behavior: the plan holds only what this interface declares, and
14 * the dry run shows nothing but the description — review happens at the
15 * definition site, not the plan.
16 *
17 * Convention: one file, one step. Steps live in `database/steps/` named
18 * `<date>-<description>Step.php` (e.g.
19 * `20261018-medias-morph-backfillStep.php`), the class defined in the same
20 * file, and the file ends with `return new MediasMorphBackfillStep();` —
21 * the command `require`s the file once and uses the returned instance.
22 *
23 * The closure's connection discipline is identical to an SQL statement
24 * step's: the run happens inside the plan's lock and transaction, and a
25 * throw vetoes the plan with the same rollback semantics a failing
26 * statement already has. Use throws as veto gates (precondition checks,
27 * row-count sanity limits) instead of dialect forks (`RAISE` / `SIGNAL`).
28 *
29 * Choose the callable form when the transform's intelligence lives in PHP:
30 * class-string constants (morph aliases), parameter-bound writes,
31 * conditionals. Choose the SQL step for pure row transforms — the
32 * visible-by-default form stays the default.
33 *
34 * @see \BlueprintAU\Radiant\Database\Schema\DataStep The plan vehicle the command adapts instances into.
35 */
36interface SyncStep
37{
38    /**
39     * The human description of the step's effects — all a dry run can
40     * show for a callable step, and the prompt line the destructive
41     * confirm gate displays.
42     */
43    public function describe(): string;
44
45    /**
46     * Whether the step can lose or rewrite existing data. Destructive
47     * steps route through the same confirm gate (`--force` bypasses it)
48     * as destructive schema changes; declare false only for insert-only
49     * transforms.
50     */
51    public function destructive(): bool;
52
53    /**
54     * The table the step primarily targets — a cosmetic display and
55     * lock-scoping label, exactly like an SQL step's dominant-table
56     * guess. Never used for dispatch.
57     */
58    public function table(): string;
59
60    /**
61     * Run the transform against the live connection.
62     *
63     * @param  SqlConnection  $conn  The plan's connection — use its
64     *      statement/parameter-binding API rather than reaching for a
65     *      second connection, so the run stays inside the plan's
66     *      transaction and lock discipline.
67     */
68    public function run(SqlConnection $conn): void;
