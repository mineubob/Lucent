9/**
10 * Opt-in route model binding for controller parameters.
11 *
12 * A parameter type-hinted to a Radiant {@see \BlueprintAU\Radiant\Model}
13 * is resolved from the URL ONLY when it carries this attribute — a bare
14 * Model type-hint is never auto-bound (the IDOR fix; the old
15 * MODEL_BINDING=implicit gate is gone).
16 *
17 * The route variable is always the parameter name. What the attribute
18 * controls is HOW the lookup key is resolved:
19 *
20 * - `resolve`: a column name (bind that column to the route variable) or a
21 *   callable returning the full `[column => value]` map (required for
22 *   composite primary keys).
23 * - `scope`: a callable applied to the query before execution — the
24 *   ownership/tenant check.
25 * - `connection`: a named connection, or a callable that returns the
26 *   connection name — the lookup runs on that connection and the previous
27 *   active connection is restored afterwards.
28 *
29 * PHP attributes only accept CONSTANT expressions, so callables must be
30 * passed as invokable class-strings (`SomeScope::class`) or
31 * `[Class::class, 'method']` arrays of constants — a closure or `fn()` in
32 * an attribute argument is a compile error. Closures remain supported when
33 * a Bind instance is constructed programmatically.
34 *
35 * Exact callable signatures (enforced by convention, not by PHP):
36 *
37 * - resolve: `callable(array $vars, ServerRequestInterface $request): array<string, mixed>`
38 * - scope:   `callable(ModelQueryBuilder $query, mixed $value, array $vars, ServerRequestInterface $request): ModelQueryBuilder`
39 * - connection: `callable(array $vars, ServerRequestInterface $request): string`
40 *
41 * ```php
42 * public function show(#[Bind] User $user): Response { ... }              // by PK
43 * public function show(#[Bind(resolve: 'slug')] User $user): Response {}  // by slug
44 * public function update(
45 *     #[Bind(scope: TenantScope::class)]
46 *     User $user,
47 * ): Response { ... }
48 * public function edit(
49 *     #[Bind(scope: [TenantScope::class, 'apply'])]
50 *     User $user,
51 * ): Response { ... }
52 * ```
53 */
54#[Attribute(Attribute::TARGET_PARAMETER)]
55final class Bind
56{
57    /**
58     * @param string|callable|null $resolve Column name, or
59     *        callable(array $vars, ServerRequestInterface $request): array<string, mixed>
60     * @param callable|null $scope
61     *        callable(ModelQueryBuilder $query, mixed $value, array $vars, ServerRequestInterface $request): ModelQueryBuilder
62     * @param string|callable|null $connection Named connection, or
63     *        callable(array $vars, ServerRequestInterface $request): string
64     */
65    public function __construct(
66        public readonly mixed $resolve = null,
67        public readonly mixed $scope = null,
68        public readonly mixed $connection = null,
69    ) {
70    }
