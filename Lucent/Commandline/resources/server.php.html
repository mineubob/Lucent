20$publicPath = getcwd();

22$uri = urldecode(

26// Path-traversal guard: reject any URI that escapes the document root.
27// urldecode of %2e%2e combined with file_exists could otherwise expose
28// files outside the docroot (e.g. .env, storage/, vendor/).
29if (str_contains($uri, '..') || str_contains($uri, '\\')) {
30    http_response_code(404);

34// Serve real static files directly. Returning `false` hands control back to
35// the built-in server, which serves the file without invoking the framework.
36// PHP logs these requests itself (e.g. `[200]: GET /style.css`).
37if ($uri !== '/' && file_exists($publicPath . $uri)) {
38    return false;

41// PHP's built-in server does NOT log requests that are handled by the router
42// (it only logs connection open/close). So we write a request line ourselves,
