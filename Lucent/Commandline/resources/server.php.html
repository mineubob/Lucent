20$uri = urldecode(

24// Path-traversal guard: reject any URI that escapes the document root.
25// urldecode of %2e%2e combined with file_exists could otherwise expose
26// files outside the docroot (e.g. .env, storage/, vendor/).
27if (str_contains($uri, '..') || str_contains($uri, '\\')) {
28    http_response_code(404);

32// Serve real static files directly. Returning `false` hands control back to
33// the built-in server, which serves the file without invoking the framework.
34// PHP logs these requests itself (e.g. `[200]: GET /style.css`).
35if ($uri !== '/' && file_exists($publicPath . $uri)) {
36    return false;
37}

39// PHP's built-in server does NOT log requests that are handled by the router
40// (it only logs connection open/close). So we write a request line ourselves,
